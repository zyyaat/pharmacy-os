   188→// UnitConversion represents unit conversion rules for a product
   189→// Example: 1 box = 5 strips, 1 strip = 10 tablets → 1 box = 50 tablets
   190→type UnitConversion struct {
   191→        ID               string    `json:"id"`

   193→        FromUnit         string    `json:"from_unit"`
   194→        ToUnit           string    `json:"to_unit"`

   196→        IsStandard       bool      `json:"is_standard"`
   197→        Description      *string   `json:"description,omitempty"`
   198→        CreatedAt        time.Time `json:"created_at"`
   199→}
